cmake_minimum_required(VERSION 3.13)

project(qbe LANGUAGES C)

# ---------------------------------------------------------------------------
# Options
# ---------------------------------------------------------------------------

# The Makefile builds a -g binary with optimizations disabled; mirror that
# unless a configuration is picked explicitly.
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
	set(CMAKE_BUILD_TYPE Debug CACHE STRING "Build type" FORCE)
endif()

option(QBE_BUILD_TESTS "Register the tests from test/ with CTest" ON)

set(QBE_DEFAULT_TARGET "" CACHE STRING
	"Backend qbe defaults to, auto-detected when empty")
set_property(CACHE QBE_DEFAULT_TARGET PROPERTY STRINGS
	"" amd64_sysv amd64_apple amd64_win arm64 arm64_apple rv64)

set(QBE_TEST_TARGET "" CACHE STRING
	"Backend the test suite runs on, host default when empty")
set_property(CACHE QBE_TEST_TARGET PROPERTY STRINGS
	"" arm64 rv64 x86_64 amd64_win)

# QBE relies on assert() for its internal checks, so never compile them out.
foreach(qbe_cfg RELEASE RELWITHDEBINFO MINSIZEREL)
	string(REPLACE "-DNDEBUG" "" CMAKE_C_FLAGS_${qbe_cfg}
		"${CMAKE_C_FLAGS_${qbe_cfg}}")
endforeach()

if(MSVC)
	message(FATAL_ERROR
		"QBE needs <unistd.h> and other POSIX headers. "
		"Build it with GCC, Clang or MinGW instead of MSVC.")
endif()

# ---------------------------------------------------------------------------
# config.h
#
# Generated here instead of by the shell snippet in the Makefile.  Only
# Deftgt is needed; the asm flavour is a property of the target now.
# ---------------------------------------------------------------------------

string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" qbe_cpu)

if(QBE_DEFAULT_TARGET)
	set(qbe_deftgt "T_${QBE_DEFAULT_TARGET}")
elseif(APPLE AND qbe_cpu MATCHES "^(aarch64|arm64)$")
	set(qbe_deftgt "T_arm64_apple")
elseif(APPLE)
	set(qbe_deftgt "T_amd64_apple")
elseif(qbe_cpu MATCHES "^(aarch64|arm64)$")
	set(qbe_deftgt "T_arm64")
elseif(qbe_cpu MATCHES "^riscv64$")
	set(qbe_deftgt "T_rv64")
else()
	set(qbe_deftgt "T_amd64_sysv")
endif()

# main.c does #include "config.h", which resolves relative to main.c first.
# A config.h left over from the Makefile build would therefore win over the
# one generated below.
if(EXISTS "${qbe_SOURCE_DIR}/config.h")
	message(WARNING
		"${qbe_SOURCE_DIR}/config.h is left over from the Makefile build and "
		"will shadow the CMake-generated one. Please delete it.")
endif()

file(WRITE "${qbe_BINARY_DIR}/config.h"
	"/* Generated by CMake, do not edit. */\n"
	"#define Deftgt ${qbe_deftgt}\n")

message(STATUS "qbe: default target ${qbe_deftgt}")

# ---------------------------------------------------------------------------
# qbe
#
# Same file list as the Makefile: every backend is compiled in and picked
# at run time with -t.
# ---------------------------------------------------------------------------

add_executable(qbe
	main.c
	util.c
	parse.c
	abi.c
	cfg.c
	mem.c
	ssa.c
	alias.c
	load.c
	copy.c
	fold.c
	gvn.c
	gcm.c
	simpl.c
	ifopt.c
	live.c
	spill.c
	rega.c
	emit.c
	amd64/targ.c
	amd64/sysv.c
	amd64/isel.c
	amd64/emit.c
	amd64/winabi.c
	arm64/targ.c
	arm64/abi.c
	arm64/isel.c
	arm64/emit.c
	rv64/targ.c
	rv64/abi.c
	rv64/isel.c
	rv64/emit.c
)

set_target_properties(qbe PROPERTIES
	C_STANDARD 99
	C_STANDARD_REQUIRED ON
	C_EXTENSIONS OFF
)

target_include_directories(qbe PRIVATE
	"${qbe_SOURCE_DIR}"
	"${qbe_BINARY_DIR}"
)

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
	target_compile_options(qbe PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(GNUInstallDirs)
install(TARGETS qbe RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")

# ---------------------------------------------------------------------------
# Tests
# ---------------------------------------------------------------------------

if(QBE_BUILD_TESTS)
	enable_testing()

	file(GLOB qbe_tests "${qbe_SOURCE_DIR}/test/[!_]*.ssa")
	list(SORT qbe_tests)

	foreach(qbe_test IN LISTS qbe_tests)
		get_filename_component(qbe_name "${qbe_test}" NAME_WE)
		add_test(NAME "${qbe_name}"
			COMMAND "${qbe_SOURCE_DIR}/tools/test.sh" "${qbe_test}")
		# test.sh honours "skip <target>" markers on its own, and returns
		# success for them.  It writes to fixed paths under /tmp, so the
		# cases must not run concurrently.
		set_tests_properties("${qbe_name}" PROPERTIES
			ENVIRONMENT "bin=$<TARGET_FILE:qbe>;TARGET=${QBE_TEST_TARGET}"
			RUN_SERIAL TRUE)
	endforeach()

	add_custom_target(check
		COMMAND "${CMAKE_CTEST_COMMAND}" --output-on-failure
		WORKING_DIRECTORY "${qbe_BINARY_DIR}"
		DEPENDS qbe
		COMMENT "Running the QBE test suite"
	)

	# Cross targets, like their Makefile counterparts; each needs the
	# matching cross compiler and qemu in PATH.
	foreach(qbe_tgt x86_64 arm64 rv64 amd64_win)
		add_custom_target(check-${qbe_tgt}
			COMMAND "${CMAKE_COMMAND}" -E env
				"bin=$<TARGET_FILE:qbe>" "TARGET=${qbe_tgt}"
				"${qbe_SOURCE_DIR}/tools/test.sh" all
			WORKING_DIRECTORY "${qbe_SOURCE_DIR}"
			DEPENDS qbe
			COMMENT "Running the QBE test suite on ${qbe_tgt}"
		)
	endforeach()
endif()
